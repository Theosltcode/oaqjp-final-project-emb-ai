"""
Serveur Flask pour l'application de détection d'émotions.
"""
from flask import Flask, render_template, request
from EmotionDetection.emotion_detection import emotion_detector

app = Flask("Emotion Detector")

@app.route("/emotionDetector")
def sent_analyzer():
    """
    Analyse le texte fourni et gère les entrées invalides ou vides.
    """
    text_to_analyze = request.args.get('textToAnalyze')
    response = emotion_detector(text_to_analyze)
    
    # Vérification si l'émotion dominante est None (entrée invalide/vide)
    if response['dominant_emotion'] is None:
        return "Texte invalide ! Veuillez réessayer !"
    
    anger = response['anger']
    disgust = response['disgust']
    fear = response['fear']
    joy = response['joy']
    sadness = response['sadness']
    dominant_emotion = response['dominant_emotion']
    
    return (
        f"Pour l'énoncé donné, la réponse du système est "
        f"'anger': {anger}, 'disgust': {disgust}, 'fear': {fear}, "
        f"'joy': {joy} et 'sadness': {sadness}. "
        f"L'émotion dominante est {dominant_emotion}."
    )

@app.route("/")
def render_index_page():
    """
    Affiche la page d'accueil.
    """
    return render_template('index.html')

if __name__ == "__main__":
    app.run(host="0.0.0.0", port=5000)
